35 /**
36 * @ingroup range_iterators
37 * @brief Iterator for a itertools::multiplied (cartesian product) range.
38 *
39 * @details It stores three tuples of iterators of the original ranges:
40 * - `its_begin` contains the begin iterators of all ranges
41 * - `its_end` contains the end iterators of all ranges
42 * - `its` contains the current iterators of all ranges
43 *
44 * Incrementing is done from right to left, i.e. the iterator of the last range is incremented first.
45 * Once an iterator reaches the end of its range, it is reset to the beginning and the iterator of the
46 * previous range is incremented once.
47 *
48 * Dereferencing returns a tuple containing the results of dereferencing each iterator.
49 *
50 * See itertools::product(Rs &&...rgs) for more details.
51 *
52 * @tparam EndIters Tuple type containing the end iterators of all ranges.
53 * @tparam Iters Iterator types.
54 */
55 template <typename EndIters, typename... Iters>
56 struct prod_iter : iterator_facade<prod_iter<EndIters, Iters...>, std::tuple<typename std::iterator_traits<Iters>::value_type...>> {
57 /// Tuple containing the begin iterators of the original ranges.
58 std::tuple<Iters...> its_begin;
59
60 /// Tuple containing the end iterators of the original ranges.
61 EndIters its_end;
62
63 /// Tuple containing the current iterators of the original ranges.
64 std::tuple<Iters...> its = its_begin;
65
66 /// Number of original ranges.
67 static constexpr long Rank = sizeof...(Iters);
68
69 /// Default constructor.
70 prod_iter() = default;
71
72 /**
73 * @brief Construct a product iterator from given begin iterators and end iterators.
74 *
75 * @param its_begin Tuple containing begin iterators of the original ranges.
76 * @param its_end Tuple containing end iterators of the original ranges.
77 */
78 prod_iter(std::tuple<Iters...> its_begin, EndIters its_end) : its_begin(std::move(its_begin)), its_end(std::move(its_end)) {}
79
80 private:
81 // Helper function to recursively increment the current iterators.
82 template <int N> void _increment() {
83 // increment Nth iterator
84 ++std::get<N>(its);
85 // recursively increment previous iterators if necessary
86 if constexpr (N > 0) {
87 // if Nth iterator is at its end, reset it to its begin iterator and increment N-1st iterator
88 if (std::get<N>(its) == std::get<N>(its_end)) {
89 std::get<N>(its) = std::get<N>(its_begin);
90 _increment<N - 1>();
91 }
92 }
93 }
94
95 public:
96 /// Increment the iterator by incrementing the current iterators starting with the iterator of the last range.
97 void increment() { _increment<Rank - 1>(); }
98
99 /**
100 * @brief Equal-to operator for two itertools::prod_iter objects.
101 *
102 * @param other itertools::prod_iter to compare with.
103 * @return True, if all original iterators are equal.
104 */
105 [[nodiscard]] bool operator==(prod_iter const &other) const { return its == other.its; }
106
107 /**
108 * @brief Equal-to operator for a itertools::prod_iter and an itertools::sentinel_t.
109 *
110 * @details We reach the end of the product range, when the first iterator, i.e. `std::get<0>(its)`, is at its end.
111 *
112 * @tparam SentinelIter Iterator type of the sentinel.
113 * @param s itertools::sentinel_t to compare with.
114 * @return True, if the first iterator, i.e. `std::get<0>(its)`, is equal to the iterator of the sentinel.
115 */
116 template <typename SentinelIter> [[nodiscard]] bool operator==(sentinel_t<SentinelIter> const &s) const { return (s.it == std::get<0>(its)); }
117
118 private:
119 // Helper function to dereference all original iterators.
120 template <size_t... Is> [[gnu::always_inline]] [[nodiscard]] auto tuple_map_impl(std::index_sequence<Is...>) const {
121 return std::tuple<decltype(*std::get<Is>(its))...>(*std::get<Is>(its)...);
122 }
123
124 public:
125 /**
126 * @brief Dereference the iterator.
127 * @return Tuple containing the dereferenced values of all original iterators.
128 */
129 [[nodiscard]] decltype(auto) dereference() const { return tuple_map_impl(std::index_sequence_for<Iters...>{}); }
130 };

132 /**
133 * @ingroup adapted_ranges
134 * @brief Represents a cartesian product of ranges.
135 *
136 * @details See itertools::product(Rs &&...rgs) for more details.
137 *
138 * @tparam Rs Range types.
139 */
140 template <typename... Rs> struct multiplied {
141 /// Tuple containing the original ranges.
142 std::tuple<Rs...> tu;
143
144 /// Iterator type of the product range.
145 using iterator = prod_iter<std::tuple<decltype(std::end(std::declval<Rs &>()))...>, decltype(std::begin(std::declval<Rs &>()))...>;
146
147 /// Const iterator type the product range.
148 using const_iterator = prod_iter<std::tuple<decltype(std::cend(std::declval<Rs &>()))...>, decltype(std::cbegin(std::declval<Rs &>()))...>;
149
150 /**
151 * @brief Constructs a cartesian product (multiplied) range from the given ranges.
152 *
153 * @tparam Us Range types.
154 * @param rgs Ranges to be multiplied.
155 */
156 template <typename... Us> multiplied(Us &&...rgs) : tu{std::forward<Us>(rgs)...} {}
157
158 /// Default equal-to operator.
159 [[nodiscard]] bool operator==(multiplied const &) const = default;
160
161 private:
162 // Helper function to create a itertools::prod_iter representing the beginning of the product range.
163 template <size_t... Is> [[gnu::always_inline]] auto _begin(std::index_sequence<Is...>) {
164 return iterator{std::make_tuple(std::begin(std::get<Is>(tu))...), std::make_tuple(std::end(std::get<Is>(tu))...)};
165 }
166
167 // Const version of _begin(std::index_sequence<Is...>).
168 template <size_t... Is> [[gnu::always_inline]] auto _cbegin(std::index_sequence<Is...>) const {
169 return const_iterator{std::make_tuple(std::cbegin(std::get<Is>(tu))...), std::make_tuple(std::cend(std::get<Is>(tu))...)};
170 }
171
172 public:
173 /**
174 * @brief Beginning of the product range.
175 * @return itertools::prod_iter representing the beginning of the product range.
176 */
177 [[nodiscard]] iterator begin() noexcept { return _begin(std::index_sequence_for<Rs...>{}); }
178
179 /// Const version of begin().
180 [[nodiscard]] const_iterator cbegin() const noexcept { return _cbegin(std::index_sequence_for<Rs...>{}); }
181
182 /// Const overload of begin().
183 [[nodiscard]] const_iterator begin() const noexcept { return cbegin(); }
184
185 /**
186 * @brief End of the product range.
187 * @return itertools::sentinel_t containing the end iterator of the first original range, i.e. `std::end(std::get<0>(tu))`.
188 */
189 [[nodiscard]] auto end() noexcept { return make_sentinel(std::end(std::get<0>(tu))); }
190
191 /// Const version of end().
192 [[nodiscard]] auto cend() const noexcept { return make_sentinel(std::cend(std::get<0>(tu))); }
193
194 /// Const overload of end().
195 [[nodiscard]] auto end() const noexcept { return cend(); }
196 };

206 /**
207 * @brief Lazy-multiply a given number of ranges by forming their cartesian product.
208 *
209 * @details An arbitrary number of ranges are multiplied together into a cartesian product range.
210 * They are traversed such that the last range is traversed the fastest (see the example below).
211 * The number of elements in a product range is equal to the product of the sizes of the given ranges.
212 * This function returns an iterable lazy object, which can be used in range-based for loops:
213 *
214 * @code{.cpp}
215 * std::vector<int> v1 { 1, 2, 3 };
216 * std::vector<char> v2 { 'a', 'b' };
217 *
218 * for (auto [i, c] : product(v1, v2)) {
219 * std::cout << "(" << i << ", " << c << ")\n";
220 * }
221 * @endcode
222 *
223 * Output:
224 *
225 * ```
226 * (1, a)
227 * (1, b)
228 * (2, a)
229 * (2, b)
230 * (3, a)
231 * (3, b)
232 * ```
233 *
234 * See also <a href="https://en.cppreference.com/w/cpp/ranges/cartesian_product_view">std::ranges::views::cartesian_product</a>.
235 *
236 * @tparam Rs Range types.
237 * @param rgs Ranges to be used.
238 * @return A product (itertools::multiplied) range.
239 */
240 template <typename... Rs> [[nodiscard]] itertools::multiplied<Rs...> product(Rs &&...rgs) { return {std::forward<Rs>(rgs)...}; }

244 // Helper function to create a product range from a container of ranges.
245 template <typename C, size_t... Is> [[gnu::always_inline]] [[nodiscard]] auto make_product_impl(C &cont, std::index_sequence<Is...>) {
246 return product(cont[Is]...);
247 }

251 /**
252 * @brief Create a cartesian product range from an array of ranges.
253 *
254 * @tparam R Range type.
255 * @tparam N Number of ranges.
256 * @param arr Array of ranges.
257 * @return A product (itertools::multiplied) range from the ranges in the array.
258 */
259 template <typename R, size_t N> [[nodiscard]] auto make_product(std::array<R, N> &arr) {
