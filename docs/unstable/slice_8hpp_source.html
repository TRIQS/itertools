34 /**
35 * @ingroup adapted_ranges
36 * @brief Represents a sliced range.
37 *
38 * @details See itertools::slice(R &&, std::ptrdiff_t, std::ptrdiff_t) for more details.
39 *
40 * @tparam R Range type.
41 */
42 template <typename R> struct sliced {
43 /// Original range.
44 R rg;
45
46 /// Index at which the sliced range starts.
47 std::ptrdiff_t start_idx;
48
49 /// Index at which the sliced range ends.
50 std::ptrdiff_t end_idx;
51
52 /// Iterator type of the sliced range.
53 using iterator = decltype(std::begin(rg));
54
55 /// Const iterator type of the sliced range.
56 using const_iterator = decltype(std::cbegin(rg));
57
58 /// Default equal-to operator.
59 [[nodiscard]] bool operator==(sliced const &) const = default;
60
61 /**
62 * @brief Helper function to calculate the size of the sliced range.
63 * @return Number of elements in the slice.
64 */
65 [[nodiscard]] std::ptrdiff_t size() const {
66 std::ptrdiff_t total_size = distance(std::cbegin(rg), std::cend(rg));
67 return std::min(total_size, end_idx) - start_idx;
68 }
69
70 /**
71 * @brief Beginning of the sliced range.
72 * @return Iterator to the beginning of the sliced range.
73 */
74 [[nodiscard]] iterator begin() noexcept { return std::next(std::begin(rg), start_idx); }
75
76 /// Const version of begin().
77 [[nodiscard]] const_iterator cbegin() const noexcept { return std::next(std::cbegin(rg), start_idx); }
78
79 /// Const overload of begin().
80 [[nodiscard]] const_iterator begin() const noexcept { return cbegin(); }
81
82 /**
83 * @brief End of the sliced range.
84 * @return Iterator to the end of the sliced range.
85 */
86 [[nodiscard]] iterator end() noexcept { return std::next(begin(), size()); }
87
88 /// Const version of end().
89 [[nodiscard]] const_iterator cend() const noexcept { return std::next(cbegin(), size()); }
90
91 /// Const overload of end().
92 [[nodiscard]] const_iterator end() const noexcept { return cend(); }
93 };

95 /**
96 * @ingroup range_adapting_functions
97 * @brief Lazy-slice a given range.
98 *
99 * @details Only the part of the given range between the `start_idx` and the `end_idx` is taken into account.
100 * If `end_idx` is bigger than the size of the original range, the slice ends at the end of the original range.
101 * If `end_idx` is smaller than `start_idx`, the slice is empty. Note that the behaviour is undefined if
102 * `start_idx` is smaller than zero.
103 *
104 * This function returns an iterable lazy object, which can be used in range-based for loops:
105 *
106 * @code{.cpp}
107 * std::array<int, 5> arr { 1, 2, 3, 4, 5 };
108 *
109 * for (auto i : slice(arr, 1, 3)) {
110 * std::cout << i << " ";
111 * }
112 * std::cout << "\n";
113 *
114 * for (auto i : slice(arr, 3, 7)) {
115 * std::cout << i << " ";
116 * }
117 * std::cout << "\n";
118 *
119 * for (auto i : slice(arr, 4, 3)) {
120 * std::cout << i << " "; // empty slice
121 * }
122 * @endcode
123 *
124 * Output:
125 *
126 * ```
127 * 2 3
128 * 4 5
129 * ```
130 *
131 * @tparam R Range type.
132 * @param rg Range to be sliced.
133 * @param start_idx Index where the slice starts.
134 * @param end_idx Index of the first element past the end of the sliced range (excluded).
135 * @return A itertools::sliced range.
136 */
137 template <typename R> [[nodiscard]] sliced<R> slice(R &&rg, std::ptrdiff_t start_idx, std::ptrdiff_t end_idx) {
138 return {std::forward<R>(rg), start_idx, std::max(start_idx, end_idx)};
139 }
