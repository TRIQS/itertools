34 /**
35 * @ingroup range_iterators
36 * @brief Iterator for an itertools::enumerated range.
37 *
38 * @details It stores an iterator of the original range and an index. Incrementing advances the iterator
39 * and the index by 1. Dereferencing returns a std::pair consisting of the current index and the current
40 * dereferenced value of the original iterator.
41 *
42 * See itertools::enumerate(R &&) for more details.
43 *
44 * @tparam Iter Iterator type.
45 */
46 template <typename Iter> struct enum_iter : iterator_facade<enum_iter<Iter>, std::pair<long, typename std::iterator_traits<Iter>::value_type>> {
47 /// Iterator of the original range.
48 Iter it;
49
50 /// Index for enumerating.
51 long i = 0;
52
53 /// Default constructor sets the index to zero and default constructs the original iterator.
54 enum_iter() = default;
55
56 /**
57 * @brief Construct an enumerated iterator from a given iterator and set the index to zero.
58 * @param it Iterator of the original range.
59 */
60 enum_iter(Iter it) : it(std::move(it)) {}
61
62 /// Increment the iterator by incrementing the original iterator and the index.
63 void increment() {
64 ++it;
65 ++i;
66 }
67
68 /**
69 * @brief Equal-to operator for two itertools::enum_iter objects.
70 *
71 * @param other itertools::enum_iter to compare with.
72 * @return True, if the original iterators are equal.
73 */
74 [[nodiscard]] bool operator==(enum_iter const &other) const { return it == other.it; }
75
76 /**
77 * @brief Equal-to operator for a itertools::enum_iter and an itertools::sentinel_t.
78 *
79 * @tparam SentinelIter Iterator type of the sentinel.
80 * @param s itertools::sentinel_t to compare with.
81 * @return True, if the original iterator is equal to the iterator stored in the sentinel.
82 */
83 template <typename SentinelIter> [[nodiscard]] bool operator==(sentinel_t<SentinelIter> const &s) const { return it == s.it; }
84
85 /**
86 * @brief Dereference the iterator.
87 * @return Tuple consisting of the current index and the current dereferenced value of the original iterator.
88 */
89 [[nodiscard]] decltype(auto) dereference() const { return std::tuple<long, decltype(*it)>{i, *it}; }
90 };

92 /**
93 * @ingroup adapted_ranges
94 * @brief Represents an enumerated range.
95 *
96 * @details See itertools::enumerate(R &&) for more details.
97 *
98 * @tparam R Range type.
99 */
100 template <typename R> struct enumerated {
101 /// Original range.
102 R rg;
103
104 /// Iterator type of the enumerated range.
105 using iterator = enum_iter<decltype(std::begin(rg))>;
106
107 /// Const iterator type of the enumerated range.
108 using const_iterator = enum_iter<decltype(std::cbegin(rg))>;
109
110 /// Default equal-to operator.
111 [[nodiscard]] bool operator==(enumerated const &) const = default;
112
113 /**
114 * @brief Beginning of the enumerated range.
115 * @return itertools::enum_iter constructed from the beginning of the original range with the index set to zero.
116 */
117 [[nodiscard]] iterator begin() noexcept { return std::begin(rg); }
118
119 /// Const version of begin().
120 [[nodiscard]] const_iterator cbegin() const noexcept { return std::cbegin(rg); }
121
122 /// Const overload of begin().
123 [[nodiscard]] const_iterator begin() const noexcept { return cbegin(); }
124
125 /**
126 * @brief End of the enumerated range.
127 * @return itertools::sentinel_t containing the end iterator of the original range.
128 */
129 [[nodiscard]] auto end() noexcept { return make_sentinel(std::end(rg)); }
130
131 /// Const version of end().
132 [[nodiscard]] auto cend() const noexcept { return make_sentinel(std::cend(rg)); }
133
134 /// Const overload of end().
135 [[nodiscard]] auto end() const noexcept { return cend(); }
136 };

138 /**
139 * @ingroup range_adapting_functions
140 * @brief Lazy-enumerate a given range (similar to Python's enumerate).
141 *
142 * @details Each element in the original range is assigned an index, starting from zero. This function
143 * returns an iterable lazy object (a itertools::enumerated range), which iterates over tuples consisting
144 * of the index and the value of the dereferenced iterator of the original range:
145 *
146 * @code{.cpp}
147 * std::vector<char> vec { 'a', 'b', 'c' };
148 *
149 * for (auto [idx, val] : enumerate(vec)) {
150 * std::cout << "(" << idx << ", " << val << ")\n";
151 * }
152 * @endcode
153 *
154 * Output:
155 *
156 * ```
157 * (0, a)
158 * (1, b)
159 * (2, c)
160 * ```
161 *
162 * See also <a href="https://en.cppreference.com/w/cpp/ranges/enumerate_view">std::ranges::views::enumerate</a>.
163 *
164 * @tparam R Range type.
165 * @param rg Range to enumerate.
166 * @return A itertools::enumerated range.
167 */
168 template <typename R> [[nodiscard]] enumerated<R> enumerate(R &&rg) { return {std::forward<R>(rg)}; }
