41 /**
42 * @brief A lazy range of integers that mimics a Python range.
43 *
44 * @details It stores the first value, the last value (excluded) and the step size between two indices.
45 * By default, the step size is set to 1.
46 *
47 * This function returns an iterable lazy object, which can be used in range-based for loops:
48 *
49 * @code{.cpp}
50 * for (auto i : range(5)) {
51 * std::cout << i << " ";
52 * }
53 * std::cout << "\n";
54 *
55 * for (auto i : range(-2, 1)) {
56 * std::cout << i << " ";
57 * }
58 * std::cout << "\n";
59 *
60 * for (auto i : range(10, 3, -2)) {
61 * std::cout << i << " ";
62 * }
63 * std::cout << "\n";
64 *
65 * for (auto i : range(0, 10, -1)) {
66 * std::cout << i << " "; // empty
67 * }
68 * @endcode
69 *
70 * Output:
71 *
72 * ```
73 * 0 1 2 3 4
74 * -2 -1 0
75 * 10 8 6 4
76 * ```
77 *
78 * See also <a href="https://en.cppreference.com/w/cpp/ranges/iota_view">std::ranges::views::iota</a>.
79 */
80 class range {
81 // First value of the range.
82 long first_ = 0;
83
84 // Last value of the range (excluded).
85 long last_ = -1;
86
87 // Number of integers between two elements of the range.
88 long step_ = 1;
89
90 public:
91 /**
92 * @ingroup integer_range
93 * @brief Denote a full range at compile-time.
94 * @details Can be used for accessing slices of multi-dimensional arrays.
95 */
96 struct all_t {};
97
98 /// See range::all_t.
99 static inline constexpr all_t all = {};
100
101 /// Integer type for backward compatibility.
102 using index_t = long;
103
104 /**
105 * @brief Default constructor.
106 * @deprecated Use range::range(long, long) or range::range(long, long, long) instead.
107 */
108 [[deprecated("range default construction deprecated. Use range::all for full range in slicing operation")]] range() = default;
109
110 /**
111 * @brief Construct a range with a step size of 1 and a given first and last (excluded) value.
112 *
113 * @param first First value of the range.
114 * @param last Last value of the range (excluded).
115 */
116 range(long first, long last) noexcept : first_(first), last_(last) {}
117
118 /**
119 * @brief Construct a range with a given step size and a given first and last (excluded) value.
120 *
121 * @details Throws an exception if the step size is zero.
122 *
123 * @param first First value of the range.
124 * @param last Last value of the range (excluded).
125 * @param step Number of integers between two elements of the range.
126 */
127 range(long first, long last, long step) : first_(first), last_(last), step_(step) {
128 if (step_ == 0) throw std::runtime_error("Step-size cannot be zero in construction of integer range");
129 }
130
131 /**
132 * @brief Construct a range with a step size of 1, a first value set to 0 and a given last value (excluded).
133 * @param last Last value of the range (excluded).
134 */
135 explicit range(long last) : range(0, last, 1) {}
136
137 /// Default equal-to operator.
138 [[nodiscard]] bool operator==(range const &) const = default;
139
140 /// Get first value of the range.
141 [[nodiscard]] long first() const { return first_; }
142
143 /// Get last value of the range (excluded).
144 [[nodiscard]] long last() const { return last_; }
145
146 /// Get step size between two elements of the range.
147 [[nodiscard]] long step() const { return step_; }
148
149 /// Get number of elements in the range.
150 [[nodiscard]] long size() const { return std::max(0l, (last_ + step_ - (step_ > 0 ? 1 : -1) - first_) / step_); }
151
152 /**
153 * @brief Shift the whole range by a given amount.
154 *
155 * @details Simply adds the given shift to the first and last value of the range, while keeping
156 * the same step size.
157 *
158 * @param shift Amount to shift the range by.
159 * @return Shifted range.
160 */
161 [[nodiscard]] range operator+(long shift) const { return {first_ + shift, last_ + shift, step_}; }
162
163 /**
164 * @brief Write the range details to std::ostream.
165 *
166 * @param os std::ostream object.
167 * @param rg range object.
168 * @return Reference to os.
169 */
170 friend inline std::ostream &operator<<(std::ostream &os, const range &rg) {
171 os << "range(" << rg.first() << "," << rg.last() << "," << rg.step() << ")";
172 return os;
173 }

175 /// Const iterator type for itertools::range.
176 struct const_iterator {
177 /// Current value.
178 long pos;
179
180 /// Last value of the range (excluded).
181 long last;
182
183 /// Step size.
184 long step;
185
186 /// Value type.
187 using value_type = long;
188
189 /// Iterator category.
190 using iterator_category = std::forward_iterator_tag;
191
192 /// Pointer type.
193 using pointer = value_type *;
194
195 /// Difference type.
196 using difference_type = std::ptrdiff_t;
197
198 /// Reference type.
199 using reference = value_type const &;
200
201 /**
202 * @brief Pre-increment operator increments the current value by the step size.
203 * @return Reference to the incremented iterator.
204 */
205 const_iterator &operator++() noexcept {
206 pos += step;
207 return *this;
208 }
209
210 /**
211 * @brief Post-increment operator increments the current value by the step size.
212 * @return Copy of the iterator before incrementing.
213 */
214 const_iterator operator++(int) noexcept {
215 const_iterator c = *this;
216 pos += step;
217 return c;
218 }
219
220 /**
221 * @brief Has the iterator reached the end of the range?
222 * @return True, if the current value of the iterator is >= the last value of the range for positive step size or
223 * if the current value of the iterator is <= the last value of the range for negative step size.
224 */
225 [[nodiscard]] bool atEnd() const noexcept { return step > 0 ? pos >= last : pos <= last; }
226
227 /**
228 * @brief Equal-to operator for two iterators.
229 *
230 * @param other Iterator to compare with.
231 * @return True, if the current values of both iterators are equal or both iterators are at the end of the range.
232 */
233 [[nodiscard]] bool operator==(const_iterator const &other) const noexcept {
234 return (other.pos == this->pos) || (other.atEnd() && this->atEnd());
235 }
236
237 /**
238 * @brief Not-equal-to operator for two iterators.
239 *
240 * @param other Iterator to compare with.
241 * @return True, if the iterators are not equal.
242 */
243 [[nodiscard]] bool operator!=(const_iterator const &other) const noexcept { return (!operator==(other)); }
244
245 /**
246 * @brief Dereference operator.
247 * @return Current value of the iterator.
248 */
249 [[nodiscard]] long operator*() const noexcept { return pos; }
250
251 /**
252 * @brief Member access operator.
253 * @return Current value of the iterator.
254 */
255 [[nodiscard]] long operator->() const noexcept { return operator*(); }
256 };
257
258 /**
259 * @brief Beginning of the integer range.
260 * @return Iterator with its current value set to the first value of the range.
261 */
262 [[nodiscard]] const_iterator cbegin() const noexcept { return {first_, last_, step_}; }
263
264 /// The same as cbegin().
265 [[nodiscard]] const_iterator begin() const noexcept { return {first_, last_, step_}; }
266
267 /**
268 * @brief End of the range.
269 * @return Iterator with its current value set to the excluded last value of the range.
270 */
271 [[nodiscard]] const_iterator cend() const noexcept { return {last_, last_, step_}; }
272
273 /// The same as cend().
274 [[nodiscard]] const_iterator end() const noexcept { return {last_, last_, step_}; }
275 };

277 /**
278 * @brief Create a cartesian product range of integer ranges from given integers.
279 *
280 * @details The given integers specify the excluded last values of the individual itertools::range objects.
281 * Each range starts at 0 and has a step size of 1.
282 *
283 * @code{.cpp}
284 * for (auto [i1, i2] : product_range(2, 3)) {
285 * std::cout << "(" << i1 << ", " << i2 << ")\n";
286 * }
287 * @endcode
288 *
289 * Output:
290 *
291 * ```
292 * (0, 0)
293 * (0, 1)
294 * (0, 2)
295 * (1, 0)
296 * (1, 1)
297 * (1, 2)
298 * ```
299 *
300 * @tparam Is Integer types.
301 * @param is Last values of the integer ranges (excluded).
302 * @return Product (itertools::multiplied) range of integer ranges. See itertools::product and itertools::range.
303 */
304 template <typename... Is, typename EnableIf = std::enable_if_t<(std::is_integral_v<Is> and ...), int>> [[nodiscard]] auto product_range(Is... is) {
305 return product(range(is)...);
306 }

317 /**
318 * @brief Create a cartesian product range of integer ranges from a tuple of integers.
319 *
320 * @details The integers in the given tuple specify the excluded last values of the individual itertools::range objects.
321 * Each range starts at 0 and has a step size of 1.
322 *
323 * @code{.cpp}
324 * for (auto [i1, i2] : product_range(std::make_tuple(2, 3))) {
325 * std::cout << "(" << i1 << ", " << i2 << ")\n";
326 * }
327 * @endcode
328 *
329 * Output:
330 *
331 * ```
332 * (0, 0)
333 * (0, 1)
334 * (0, 2)
335 * (1, 0)
336 * (1, 1)
337 * (1, 2)
338 * ```
339 *
340 * @tparam Is Integer types.
341 * @param idx_tpl Tuple containing the excluded last values of the integer ranges.
342 * @return Product (itertools::multiplied) range of integer ranges. See itertools::product and itertools::range.
343 */
344 template <typename... Is, typename EnableIf = std::enable_if_t<(std::is_integral_v<Is> and ...), int>>
345 [[nodiscard]] auto product_range(std::tuple<Is...> const &idx_tpl) {
346 return detail::product_range_impl(idx_tpl, std::make_index_sequence<sizeof...(Is)>{});
347 }

349 /**
350 * @brief Create a cartesian product range of integer ranges from an array of integers.
351 *
352 * @details The integers in the given array specify the excluded last values of the individual itertools::range objects.
353 * Each range starts at 0 and has a step size of 1.
354 *
355 * @code{.cpp}
356 * for (auto [i1, i2] : product_range(std::array{2, 3})) {
357 * std::cout << "(" << i1 << ", " << i2 << ")\n";
358 * }
359 * @endcode
360 *
361 * Output:
362 *
363 * ```
364 * (0, 0)
365 * (0, 1)
366 * (0, 2)
367 * (1, 0)
368 * (1, 1)
369 * (1, 2)
370 * ```
371 *
372 * @tparam I Integer type.
373 * @tparam N Number of elements in the array.
374 * @param idx_arr Array containing the excluded last values of the integer ranges.
375 * @return Product (itertools::multiplied) range of integer ranges. See itertools::product and itertools::range.
376 */
377 template <typename I, size_t N, typename EnableIf = std::enable_if_t<std::is_integral_v<I>, int>>
378 [[nodiscard]] auto product_range(std::array<I, N> const &idx_arr) {
379 return detail::product_range_impl(idx_arr, std::make_index_sequence<N>{});
380 }

382 /**
383 * @brief Apply a function to every element of an integer itertools::range.
384 *
385 * @code{.cpp}
386 * // print out the first 10 squares
387 * itertools::foreach(itertools::range(1, 11), [](int i) {
388 * std::cout << i * i << " ";
389 * });
390 * @endcode
391 *
392 * Output:
393 *
394 * ```
395 * 1 4 9 16 25 36 49 64 81 100
396 * ```
397 *
398 * @tparam F Callable type.
399 * @param rg itertools::range object.
400 * @param f Callable object to be applied to each element.
401 */
402 template <typename F> void foreach (range const &rg, F && f) {
403 auto i = rg.first(), last = rg.last(), step = rg.step();
404 for (; i < last; i += step) std::forward<F>(f)(i);
405 }
