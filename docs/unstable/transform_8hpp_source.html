35 /**
36 * @ingroup range_iterators
37 * @brief Iterator for a itertools::transformed range.
38 *
39 * @details It stores an iterator of the original range and a callable object that is used to transform the
40 * elements of the original range. Incrementing simply increments the iterator. Dereferencing returns the
41 * result of the callable object applied to the dereferenced iterator, i.e. the transformed element.
42 *
43 * See itertools::transform(R &&, F) for more details.
44 *
45 * @tparam Iter Iterator type.
46 * @tparam F Callable type.
47 * @tparam Value Return type of the callable.
48 */
49 template <typename Iter, typename F, typename Value = std::invoke_result_t<F, typename std::iterator_traits<Iter>::value_type>>
50 struct transform_iter : iterator_facade<transform_iter<Iter, F>, Value> {
51 /// Iterator of the original range.
52 Iter it;
53
54 /// Callable doing the transformation.
55 mutable std::optional<F> lambda;
56
57 /// Default constructor.
58 transform_iter() = default;
59
60 /**
61 * @brief Construct a transformed iterator from a given iterator and callable.
62 *
63 * @param it Iterator of the original range.
64 * @param lambda Callable doing the transformation.
65 */
66 transform_iter(Iter it, F lambda) : it(std::move(it)), lambda(std::move(lambda)) {}
67
68 /// Increment the iterator by incrementing the original iterator.
69 void increment() { ++it; }
70
71 /// Default move constructor.
72 transform_iter(transform_iter &&) = default;
73
74 /// Default copy constructor.
75 transform_iter(transform_iter const &) = default;
76
77 /// Default move assignment operator.
78 transform_iter &operator=(transform_iter &&) = default;
79
80 /**
81 * @brief Custom copy assignment operator makes sure that the optional callable is correctly copied.
82 * @param other itertools::transform_iter to copy from.
83 */
84 transform_iter &operator=(transform_iter const &other) {
85 it = other.it;
86 if (other.lambda.has_value())
87 lambda.emplace(other.lambda.value());
88 else
89 lambda.reset();
90 return *this;
91 }
92
93 /**
94 * @brief Equal-to operator for two itertools::transform_iter objects.
95 *
96 * @param other itertools::transform_iter to compare with.
97 * @return True, if the original iterators are equal.
98 */
99 [[nodiscard]] bool operator==(transform_iter const &other) const { return it == other.it; }
100
101 /**
102 * @brief Equal-to operator for a itertools::transform_iter and an itertools::sentinel_t.
103 *
104 * @tparam SentinelIter Iterator type of the sentinel.
105 * @param s itertools::sentinel_t to compare with.
106 * @return True, if the original iterator is equal to the iterator stored in the sentinel.
107 */
108 template <typename SentinelIter> [[nodiscard]] bool operator==(sentinel_t<SentinelIter> const &s) const { return (it == s.it); }
109
110 /**
111 * @brief Dereference the iterator.
112 * @return Result of the callable applied to the dereferenced iterator, i.e. the transformed value.
113 */
114 [[nodiscard]] decltype(auto) dereference() const { return (*lambda)(*it); }
115 };

117 /**
118 * @ingroup adapted_ranges
119 * @brief Represents a transformed range.
120 *
121 * @details See itertools::transform(R &&, F) for more details.
122 *
123 * @tparam R Range type.
124 * @tparam F Callable type.
125 */
126 template <typename R, typename F> struct transformed {
127 /// Original range.
128 R rg;
129
130 /// Callable doing the transformation.
131 F lambda;
132
133 /// Const iterator type of the transformed range.
134 using const_iterator = transform_iter<decltype(std::cbegin(rg)), F>;
135
136 /// Iterator type of the transformed range.
137 using iterator = const_iterator;
138
139 /**
140 * @brief Beginning of the transformed range.
141 * @return itertools::transform_iter constructed from the beginning of the original range and the callable.
142 */
143 [[nodiscard]] const_iterator cbegin() const noexcept { return {std::cbegin(rg), lambda}; }
144
145 /// The same as cbegin().
146 [[nodiscard]] const_iterator begin() const noexcept { return cbegin(); }
147
148 /**
149 * @brief End of the transformed range.
150 * @return itertools::sentinel_t containing the end iterator of the original range.
151 */
152 [[nodiscard]] auto cend() const noexcept { return make_sentinel(std::cend(rg)); }
153
154 /// The same as cend().
155 [[nodiscard]] auto end() const noexcept { return cend(); }
156 };

158 /**
159 * @ingroup range_adapting_functions
160 * @brief Lazy-transform a given range by applying a unary callable object to every element of the original range.
161 *
162 * @details The value type of the transformed range depends on the return type of the callable.
163 *
164 * This function returns an iterable lazy object (a itertools::transformed range), which can be used in range-based for loops:
165 *
166 * @code{.cpp}
167 * std::list<int> list { 1, 2, 3, 4, 5 };
168 *
169 * for (auto i : itertools::transform(list, [](int i) { return i * i; })) {
170 * std::cout << i << " ";
171 * }
172 * @endcode
173 *
174 * Output:
175 *
176 * ```
177 * 1 4 9 16 25
178 * ```
179 *
180 * See also <a href="https://en.cppreference.com/w/cpp/ranges/transform_view">std::ranges::views::transform</a>.
181 *
182 * @tparam R Range type.
183 * @tparam F Callable type.
184 * @param rg Range to transform.
185 * @param lambda Callable to be applied to the given range.
186 * @return A itertools::transformed range.
187 */
188 template <typename R, typename F> [[nodiscard]] auto transform(R &&rg, F lambda) {
189 return transformed<R, F>{std::forward<R>(rg), std::move(lambda)};
190 }
