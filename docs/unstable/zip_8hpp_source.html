35 /**
36 * @ingroup range_iterators
37 * @brief Iterator for a itertools::zipped range.
38 *
39 * @details It stores iterators of the original ranges in a tuple. Incrementing simply increments each iterator
40 * individually. Dereferencing returns a tuple containing the results of dereferencing each iterator.
41 *
42 * See itertools::zip(Rs &&...rgs) for more details.
43 *
44 * @tparam Iters Iterator types.
45 */
46 template <typename... Iters>
47 struct zip_iter : iterator_facade<zip_iter<Iters...>, std::tuple<typename std::iterator_traits<Iters>::value_type...>> {
48 /// Tuple containing iterators of the original ranges.
49 std::tuple<Iters...> its;
50
51 /// Default constructor.
52 zip_iter() = default;
53
54 /**
55 * @brief Constructs a zipped iterator from given iterators.
56 * @param its Tuple containing iterators of the original ranges.
57 */
58 zip_iter(std::tuple<Iters...> its) : its(std::move(its)) {}
59
60 private:
61 // Helper function which increments all original iterators.
62 template <size_t... Is> [[gnu::always_inline]] void increment_all(std::index_sequence<Is...>) { ((void)(++std::get<Is>(its)), ...); }
63
64 public:
65 /// Increment the iterator by incrementing all original iterators stored in the tuple.
66 void increment() { increment_all(std::index_sequence_for<Iters...>{}); }
67
68 /**
69 * @brief Equal-to operator for two itertools::zip_iter objects.
70 *
71 * @param other itertools::zip_iter to compare with.
72 * @return True, if all original iterators are equal.
73 */
74 [[nodiscard]] bool operator==(zip_iter const &other) const { return its == other.its; }
75
76 /**
77 * @brief Equal-to operator for a itertools::zip_iter and an itertools::sentinel_t.
78 *
79 * @details Only one of the iterators has to be equal to the corresponding iterator of the sentinel. In case
80 * the original ranges have different lengths, the itertools::zipped range should have the length of the shortest range.
81 *
82 * @tparam SentinelIter Iterator type of the sentinel.
83 * @param s itertools::sentinel_t to compare with.
84 * @return True, if one of the original iterators is equal to the corresponding iterator of the sentinel.
85 */
86 template <typename SentinelIter> [[nodiscard]] bool operator==(sentinel_t<SentinelIter> const &s) const {
87 return [&]<size_t... Is>(std::index_sequence<Is...>) {
88 return ((std::get<Is>(its) == std::get<Is>(s.it)) || ...);
89 }(std::index_sequence_for<Iters...>{});
90 }
91
92 private:
93 // Helper function to dereference all original iterators.
94 template <size_t... Is> [[nodiscard]] auto tuple_map_impl(std::index_sequence<Is...>) const {
95 return std::tuple<decltype(*std::get<Is>(its))...>(*std::get<Is>(its)...);
96 }
97
98 public:
99 /**
100 * @brief Dereference the iterator.
101 * @return Tuple containing the dereferenced values of all original iterators.
102 */
103 [[nodiscard]] decltype(auto) dereference() const { return tuple_map_impl(std::index_sequence_for<Iters...>{}); }
104 };

106 /**
107 * @ingroup adapted_ranges
108 * @brief Represents a zipped range.
109 *
110 * @details See itertools::zip(Rs &&...rgs) for more details.
111 *
112 * @tparam Rs Range types.
113 */
114 template <typename... Rs> struct zipped {
115 /// Tuple containing the original ranges.
116 std::tuple<Rs...> tu;
117
118 /// Convenience typedef for an std::index_sequence.
119 using seq_t = std::index_sequence_for<Rs...>;
120
121 /// Iterator type of the zipped range.
122 using iterator = zip_iter<decltype(std::begin(std::declval<Rs &>()))...>;
123
124 /// Const iterator type of the zipped range.
125 using const_iterator = zip_iter<decltype(std::cbegin(std::declval<Rs &>()))...>;
126
127 /**
128 * @brief Construct a zipped range from the given ranges.
129 *
130 * @tparam Us Range types.
131 * @param rgs Ranges to zip.
132 */
133 template <typename... Us> zipped(Us &&...rgs) : tu{std::forward<Us>(rgs)...} {}
134
135 /// Default equal-to operator.
136 [[nodiscard]] bool operator==(zipped const &) const = default;
137
138 private:
139 // Helper function that applies a given callable to each range in the stored tuple and returns a new tuple with the results.
140 template <typename F, size_t... Is> [[gnu::always_inline]] auto tuple_map(F &&f, std::index_sequence<Is...>) {
141 return std::make_tuple(std::forward<F>(f)(std::get<Is>(tu))...);
142 }
143
144 // Const overload of tuple_map(F &&, std::index_sequence<Is...>).
145 template <typename F, size_t... Is> [[gnu::always_inline]] auto tuple_map(F &&f, std::index_sequence<Is...>) const {
146 return std::make_tuple(std::forward<F>(f)(std::get<Is>(tu))...);
147 }
148
149 public:
150 /**
151 * @brief Beginning of the zipped range.
152 * @return itertools::zip_iter constructed from the begin iterators of the original ranges.
153 */
154 [[nodiscard]] iterator begin() noexcept {
155 return tuple_map([](auto &&rg) { return std::begin(rg); }, seq_t{});
156 }
157
158 /// Const version of begin().
159 [[nodiscard]] const_iterator cbegin() const noexcept {
160 return tuple_map([](auto &&rg) { return std::cbegin(rg); }, seq_t{});
161 }
162
163 /// Const overload of begin().
164 [[nodiscard]] const_iterator begin() const noexcept { return cbegin(); }
165
166 /**
167 * @brief End of the zipped range.
168 * @return itertools::sentinel_t containing a tuple of end iterators of the original ranges.
169 */
170 [[nodiscard]] auto end() noexcept {
171 return make_sentinel(tuple_map([](auto &&rg) { return std::end(rg); }, seq_t{}));
172 }
173
174 /// Const version of end().
175 [[nodiscard]] auto cend() const noexcept {
176 return make_sentinel(tuple_map([](auto &&rg) { return std::cend(rg); }, seq_t{}));
177 }
178
179 /// Const overload of end().
180 [[nodiscard]] auto end() const noexcept { return cend(); }
181 };

183 /**
184 * @ingroup range_adapting_functions
185 * @brief Lazy-zip ranges together (similar to Python's zip).
186 *
187 * @details An arbitrary number of ranges are zipped together into a tuple. The zipped range will have as many
188 * elements as the shortest given range.
189 *
190 * This function returns an iterable lazy object, which can be used in range-based for loops:
191 *
192 * @code{.cpp}
193 * std::vector<int> v1 { 1, 2, 3 };
194 * std::vector<char> v2 { 'a', 'b', 'c', 'd', 'e' };
195 *
196 * for (auto [i1, i2] : zip(v1, v1)) {
197 * std::cout << "(" << i1 << ", " << i2 << ") ";
198 * }
199 *
200 * for (auto [i1, i2, c3] : zip(v1, v1, v2)) {
201 * std::cout << "(" << i1 << ", " << i2 << ", " << c3 << ") ";
202 * }
203 * @endcode
204 *
205 * Output:
206 *
207 * ```
208 * (1, 1) (2, 2) (3, 3)
209 * (1, 1, a) (2, 2, b) (3, 3, c)
210 * ```
211 *
212 * See also <a href="https://en.cppreference.com/w/cpp/ranges/zip_view">std::ranges::views::zip</a>.
213 *
214 * @tparam Rs Range types.
215 * @param rgs Ranges to zip.
216 * @return A itertools::zipped range.
217 */
218 template <typename... Rs> [[nodiscard]] zipped<Rs...> zip(Rs &&...rgs) { return {std::forward<Rs>(rgs)...}; }
